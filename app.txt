from fastapi import FastAPI, HTTPException
from pydantic import BaseModel, Field
import joblib
import pandas as pd

app = FastAPI(title="F1 Pit Stop Prediction Engine", version="1.0.0")

model = joblib.load('artifacts/f1_pit_pipeline.joblib')

class TelemetryPayload(BaseModel):
    Driver: str = Field(..., example="VER")
    Compound: str = Field(..., example="MEDIUM")
    Race: str = Field(..., example="Monaco GP")
    LapNumber: int = Field(..., ge=1, le=80, example=24)
    TyreLife: int = Field(..., ge=0, example=18)
    Stint: int = Field(..., ge=1, example=1)
    Position: int = Field(..., ge=1, le=20, example=1)
    RaceProgress: float = Field(..., ge=0.0, le=1.0, example=0.32)
    LapTime_s: float = Field(..., alias="LapTime (s)", example=74.32)
    Cumulative_Degradation: float = Field(..., example=1.45)
    LapTime_Delta: float = Field(..., example=0.28)

@app.post("/predict")
def predict_pit(payload: TelemetryPayload):
    try:
        input_data = pd.DataFrame([payload.model_dump(by_alias=True)])
        prob = float(model.predict_proba(input_data)[0, 1])
        return {
            "pit_next_lap_probability": round(prob, 4),
            "recommendation": "BOX" if prob >= 0.50 else "STAY_OUT"
        }
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
